if (!newBlock.get(myLocation.getID())) // block already taken - get synchronized
        return currentBlock; // stay where you are
// only one thread can try move to newBlock at any time
// passed previous checks so valid to move
synchronized (newBlock) {
    if (currentBlock.equals(entrance)) {
        synchronized (entrance) {
            // only release the block when you have the entrances lock (otherwise you could release the entrance and another thread obtain that lock and then try obtain this lock). This way, guaranteed to get the entrance lock (entrance still taken so no other thread can take the entrance lock)
            currentBlock.release();
            entrance.notify();
        }
    } else {
        // release it otherwise
        currentBlock.release();
    }

    // this thread won the newBlock
    myLocation.setLocation(newBlock); // setting to new block and returning
    return newBlock;
}
