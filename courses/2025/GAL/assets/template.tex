\begin{enumerate}[label={1.}, left=0pt]
  \item Let $f(x)=x^{2}+2\in\mathbb{Z}_{5}[x]$.
  \begin{enumerate}[label={(\alph*)}, left=0pt]
    \item \fbox{4 marks} Prove that $f$ is irreducible over $\mathbb{Z}_{5}$.
    \item \fbox{1 mark} How many elements does $K=\mathbb{Z}_{5}[x]/\langle f\rangle$ have?
    \item \fbox{10 marks} Let $t=x+\langle f\rangle\in K$. Show that $1+t$ is a generator of $K^{*}$. 
  \end{enumerate}
\end{enumerate}
\textit{\textbf{My Solution:}}
  \begin{enumerate}[label={(\alph*)}, left=16pt]
    \item The degree of $f$ is 2, so if $f$ is reducible over $\mathbb{Z}_5$, it must factorise into linear factors. But \begin{align*}
    f(0) & = 0^2 + 2 = 2 \\ 
    f(1) & = 1^2 + 2 = 3 \\ 
    f(2) & = 2^2 + 2 = 6 = 1 \\ 
    f(3) & = 3^2 + 2 = 11 = 1 \\ 
    f(4) & = 4^2 + 2 = 18 = 3 
    \end{align*}
    and so $f$ has no roots in $\mathbb{Z}_5$ and thus cannot be reduced into linear factors and is thus irreducible.
    \item By the Description of Polynomial Factor rings, since $f$ is a monic polynomial in $\mathbb{Z}_5[x]$ of degree $2$, \[
    \mathbb{Z}_5 [x] / \langle f \rangle = \{a_0 +a_1 t : a_0, a_1 \in \mathbb{Z}_5, f(t) = 0\}
    \]
    Then $f(t) = 0 \iff t^2 + 2 = 0 \iff t^2 = -2 = 3$ and there are $5 \times 5 = 25$ elements.
    \item $K^*$ is the multiplicative group of non-zero elements of $K$ and thus has $24$ elements. If $0_K \notin \langle 1+t \rangle$, then $\langle 1+t \rangle$ is a subgroup of $K^*$. Since $f$ is irreducible, we know that $K$ is a field. This means it contains no zero divisors, so since $1+t \ne 0$, $(1+t)^k \ne 0$ for all $k \in \mathbb{Z}$ and so $0_K \notin \langle 1+t \rangle$. By Lagrange's Theorem, $o(1+t) | 24$. So \\ $o(1+t) \in \{1,2,3,4,6,8,12,24\}$. Checking powers: \begin{align*}
      (1+t)^1 &= 1+t \ne 1 \\ 
      (1+t)^2 &= 1+2t+t^2 =4 + 2t \ne 1 \\ 
      (1+t)^3 &= (4+2t)(1+t) = 4 + 6t + 2t^2 = 10 + 6t = t \ne 1 \\ 
      (1+t)^4 &= (4+2t)^2 = 16 + 16t + 4t^2 = 28 + 16t = 3 + t \ne 1 \\ 
      (1+t)^6 &= (t)^2 = 3 \ne 1 \\ 
      (1+t)^8 &= (3+t)^2 = 9 + 6t + t^2 = 2 + t \ne 1 \\ 
      (1+t)^{12} &= (3)^2 = 9 = 4 \ne 1 \\ 
      (1+t)^{24} &= (4)^2 = 16 = 1 
    \end{align*}
    So $o(1+t) = 24$ and $\langle 1+t \rangle = K^*$.
  \end{enumerate}

\begin{enumerate}[resume, label={2.}, left=0pt]
  \item \fbox{10 marks} Let $F$ be a field of characteristic $0$ and $E:F$ a normal extension. Suppose that the Galois group $\mathrm{Gal}(E:F)$ is a cyclic group of order $30$.\\ [4pt]
  Prove that for every positive integer $d$ dividing $30$, there exists a unique intermediate field $K_{d}$ of the extension $E:F$ satisfying that $[K_{d}:F]=d$.
\end{enumerate}
\textit{\textbf{My Solution:}}
\begin{enumerate}[label={}, left=12pt]
    \item First, since Since $F$ has characteristic $0$ and $E:F$ is normal, $E:F$ is a finite Galois extension. Let positive integer $d \ | \ 30$. By The Fundamental Theorem of Finite Cyclic Groups, since $\text{Gal}(E:F)$ is a cyclic group of order $30$, there is an element of order $30/d$. Let $\sigma$ be that element. Then $|\langle \sigma \rangle | = 30/d$. Let $K_d^* = \langle \sigma \rangle$. By The Fundamental Theorem of Galois Theory, there is some intermediate field $K_d$ of $E:F$ corresponding to $K_d^*$. Moreover, we have that  \[
    [K_d : F] = |\text{Gal}(E:F) : K_d^*| = \frac{|\text{Gal}(E:F)|}{|K_d^*|} = \frac{30}{30/d} = d
    \]
    All we have to show is that this intermediate field is unique. Let $K$ be an intermediate field of the extension $E:F$ such that $[K:F] = d$. Again, by the Fundamental Theorem of Galois Theory, $|\text{Gal}(E:F):K^*| = d$ and so $|K^*| = 30/d$. But $\text{Gal}(E:F)$ is a finite cyclic group, the Fundamental Theorem of Finite Cyclic Groups ensures that there is only one subgroup with order $30/d$. That is, $K^* = K_d^*$. The map $K \mapsto K^*$ is bijective (by the Fundamental Theorem of Galois Theory) so $K = K_d$.
\end{enumerate}

\begin{enumerate}[resume, label={3.}, left=0pt]
  \item Let $E:\mathbb{Q}$ be a field extension, where $E:=\mathbb{Q}(\sqrt{2},\sqrt{3},\sqrt{5})$.
  \begin{enumerate}[label={(\alph*)}, left=0pt]
    \item \fbox{5 marks} Find the degree $[E:\mathbb{Q}]$ of the extension $E:\mathbb{Q}$.
    \item \fbox{5 marks} Give a basis of $E$ as a vector space over $\mathbb{Q}$.
    \item \fbox{10 marks} List the elements of the Galois group $\mathrm{Gal}(E:\mathbb{Q})$. To which (well known) group is $\mathrm{Gal}(E:\mathbb{Q})$ isomorphic?
    \item \fbox{10 marks} For the intermediate fields $K_{i}$ of the extension $E:\mathbb{Q}$ given below, determine the subgroup $K_{i}^{*}$ of $\mathrm{Gal}(E:\mathbb{Q})$.
    \[
      K_{1} = \mathbb{Q}(\sqrt{10}),\quad \quad K_{2} = \mathbb{Q}(\sqrt{2}+\sqrt{5}).
    \]
    \item \fbox{10 marks} How many intermediate fields $K$ of the extension $E:\mathbb{Q}$ are such that $[K:\mathbb{Q}]=4$?
    \item \fbox{15 marks} Show that if $f(x)\in\mathbb{Q}[x]$ is an irreducible (over $\mathbb{Q}$) monic polynomial of degree $3$, then $f(x)$ has no roots in $E$.
  \end{enumerate}
\end{enumerate}
\normalsize
\textit{\textbf{My Solution:}}
  \begin{enumerate}[label={(\alph*)}, left=16pt]
    \item Consider the following extensions: \[
    \mathbb{Q} \subseteq \mathbb{Q}(\sqrt{2}) \subseteq \mathbb{Q}(\sqrt{2}, \sqrt{3}) \subseteq E
    \]By the degrees law, \[
    [E: \mathbb{Q}] = [\mathbb{Q}(\sqrt{2}, \sqrt{3}, \sqrt{5}), \mathbb{Q}(\sqrt{2}, \sqrt{3})] \cdot [\mathbb{Q}(\sqrt{2}, \sqrt{3}), \mathbb{Q}(\sqrt{2})] \cdot [\mathbb{Q}(\sqrt{2}), \mathbb{Q}]
    \]
    Clearly $\text{irr}(\sqrt{2}, \mathbb{Q}) = x^2 - 2$ (irreducible by Eisenstein's Criterion). This means\\  $[\mathbb{Q}(\sqrt{2}): \mathbb{Q}] = 2$ and a basis of $\mathbb{Q}(\sqrt{2})$ over $\mathbb{Q}$ is $\{1, \sqrt{2}\}$. Now $\text{irr}(\sqrt{3}, \mathbb{Q}) = x^2 - 3$ (irreducible by Eisenstein's Criterion). But then if $x^2 - 3$ is reducible in $\mathbb{Q}(\sqrt{2})$, then $\text{deg}(\sqrt{3}, \mathbb{Q}(\sqrt{2})) = 1$ and this implies $\sqrt{3} \in \mathbb{Q}(\sqrt{2})$. So, if we can show $\sqrt{3} \notin \mathbb{Q}(\sqrt{2})$ then $\text{irr}(\sqrt{3}, \mathbb{Q}(\sqrt{2})) = x^2 - 3$. If it were, then \[
    \sqrt{3} = q_0 + q_1 \sqrt{2}
    \]
    Using that $\{1, \sqrt{2}\}$ is a basis of $\mathbb{Q}(\sqrt{2})$ over $\mathbb{Q}$.
    But $\sqrt{3}$ is not a rational multiple of $\sqrt{2}$ and is not rational, so $q_0,q_1 \ne 0$. This implies
    \[
    3 = q_0^2 + 2q_1^2 + 2q_0q_1\sqrt{2}
    \]
    But this implies $\sqrt{2}$ is rational, a contradiction. Thus $[\mathbb{Q}(\sqrt{2}, \sqrt{3}), \mathbb{Q}(\sqrt{2})] = 2$ and a basis of $\mathbb{Q}(\sqrt{2}, \sqrt{3})$ over $\mathbb{Q}(\sqrt{2})$ is $\{1,\sqrt{3}\}$. ~\\Finally, $\text{irr}(\mathbb{Q}(\sqrt{5}), \mathbb{Q}) = x^2 - 5$ (again, irreducible by Eisenstein's Criterion), so, by a similar argument as above, if we can show that $\sqrt{5} \notin \mathbb{Q}(\sqrt{2}, \sqrt{3})$, then \\ $[\mathbb{Q}(\sqrt{2}, \sqrt{3}, \sqrt{5}):\mathbb{Q}(\sqrt{2}, \sqrt{3})] = 2$. We know, by the above results, a basis of $\mathbb{Q}(\sqrt{2}, \sqrt{3})$ over $\mathbb{Q}$ is $\{1,\sqrt{2}, \sqrt{3}, \sqrt{6}\}$. Assume $\sqrt{5} \in \mathbb{Q}(\sqrt{2}, \sqrt{3})$. Then \[
    \sqrt{5} = a_0 + a_1\sqrt{2} + a_2 \sqrt{3} + a_3 \sqrt{6}
    \]
    But this implies \[
    5 = a_0^2 + 2a_1^2 + 3a_2^2 + 6a_3^2 + 2(a_0a_1 \sqrt{2} + a_0a_2\sqrt{3} + a_0a_3 \sqrt{6} + a_1a_2\sqrt{6} + a_1a_3\sqrt{12} +a_2a_3 \sqrt{18})
    \]
    And this implies \[
    (5-a_0^2 - 2a_1^2 - 3a_2^2 - 6a_3^2) + (2a_0a_1+3a_2a_3)\sqrt{2} + (2a_0a_2+2a_1a_3)\sqrt{3} + (2a_0a_3 + 2a_1a_2)\sqrt{6} = 0
    \]
    But the basis $\{1,\sqrt{2}, \sqrt{3}, \sqrt{6}\}$ is linearly independent, so each coefficient must be 0. Solving this system of equations, we need that \begin{align}
      a_0a_2 & = -a_1a_3 \\ 
      a_0a_3 & = -a_1a_2 \\ 
      2a_0a_1 & = -3a_2a_3 \\ 
      5 & = a_0^2 + 2a_1^2 + 3a_2^2 + 6a_3^2
    \end{align}
    Mulitiply $(1)$ and $(2)$. Then \[
    a_0^2 a_2a_3 = a_1^2 a_2 a_3 \iff a_2a_3 (a_0^2 - a_1^2) = a_2a_3 (a_0 - a_1)(a_0 + a_1) = 0
    \]
    From here we consider cases:
    \begin{itemize}
      \item[-] $a_2 = 0$. Then $a_1a_3 = a_0a_3 = a_0a_1 = 0$. This implies at least 2 of $a_0, a_1, a_3$ is zero. If all are zero, equation $(4)$ is violated. If one is nonzero, then, by $(4)$, it implies that element is irrational, another contradiction. So this case is impossible.
      \item[-] $a_0 = a_1$. Again, by a similar argument to above, $a_1 \ne 0$. So, by $(1)$, $a_2 = -a_3$. By $(3)$, $2a_1^2 = 3a_2^2$. But then, by $(4)$, $5 = a_1^2 + 2a_1^2 + 2a_1^2 + 4a_1^2$. That is, $5 = 9a_1^2$, implying $a_1$ is irrational, a contradiction.
      \item[-] $a_0 = -a_1$. Again, by a similar argument to above, $a_1 \ne 0$. So, by $(1)$, $a_2 = a_3$. By $(3)$, $2a_1^2 = 3a_2^2$. Then simply proceed as the above case to get a contradiction.
    \end{itemize}
    So $\sqrt{5} \notin \mathbb{Q}(\sqrt{2}, \sqrt{3})$. Finally, $[\mathbb{Q}(\sqrt{2}, \sqrt{3}, \sqrt{5}) : \mathbb{Q}(\sqrt{2}, \sqrt{3})] = 2$ (and a basis of $\mathbb{Q}(\sqrt{2}, \sqrt{3}, \sqrt{5})$ over $\mathbb{Q}(\sqrt{2}, \sqrt{3})$ is $\{1, \sqrt{5}\}$). Finally, \[
    [E:\mathbb{Q}] = 2^3 = 8
    \]
    \item In $(a)$, we have found a basis of $\mathbb{Q}(\sqrt{2}, \sqrt{3}, \sqrt{5})$ over $\mathbb{Q}(\sqrt{2}, \sqrt{3})$ (being $\{1, \sqrt{5}\}$) and a basis of $\mathbb{Q}(\sqrt{2}, \sqrt{3})$ over $\mathbb{Q}$ (being $\{1,\sqrt{2}, \sqrt{3}, \sqrt{6}\}$). Combining these, using the proof of the Degree's Law, a basis of $E$ over $\mathbb{Q}$ is \[
    \{1, \sqrt{2}, \sqrt{3}, \sqrt{6}, \sqrt{5}, \sqrt{10}, \sqrt{15}, \sqrt{30}\}
    \]
    \item We have that \begin{align*}
      \text{irr}(\sqrt{2}, \mathbb{Q}) & = x^2 - 2 \\ 
      \text{irr}(\sqrt{3}, \mathbb{Q}) & = x^2 - 3 \\ 
      \text{irr}(\sqrt{5}, \mathbb{Q}) & = x^2 - 5 
    \end{align*}
    (All irreducible by Eisenstein's Criterion). For any $\sigma \in \text{Gal}(E:\mathbb{Q})$, we must have that $\sigma(\sqrt{2})$ is a root of $x^2 - 2$. The roots of $x^2 - 2$ are $\pm \sqrt{2}$. Similarly, $\sigma(\sqrt{3}) = \pm \sqrt{3}$ and $\sigma (\sqrt{5}) = \pm \sqrt{5}$, and $\sigma$ is uniquely determined by $\sigma(\sqrt{2}), \sigma(\sqrt{3}), \sigma(\sqrt{5})$. So $\text{Gal}(E:\mathbb{Q})$ has 8 elements. Define the automorphisms \[
    \sqrt{2}  \xmapsto{\sigma_1} - \sqrt{2}, \quad \sqrt{2}  \xmapsto{\sigma_2}  \sqrt{2}, \quad \sqrt{2}  \xmapsto{\sigma_3} \sqrt{2}
    \]
    \[
    \sqrt{3}  \xmapsto{\sigma_1} \sqrt{3}, \quad \sqrt{3}  \xmapsto{\sigma_2}  -\sqrt{3}, \quad \sqrt{3}  \xmapsto{\sigma_3} \sqrt{3}
    \]
    \[
    \sqrt{5}  \xmapsto{\sigma_1} \sqrt{5}, \quad \sqrt{5}  \xmapsto{\sigma_2}  \sqrt{5}, \quad \sqrt{5}  \xmapsto{\sigma_3} -\sqrt{5}
    \]
    And $\sigma_1, \sigma_2, \sigma_3$ all fix $\mathbb{Q}$. Then \[
    \text{Gal}(E:\mathbb{Q}) = \{\text{Id}, \sigma_1, \sigma_2, \sigma_3, \sigma_1\sigma_2, \sigma_1\sigma_3, \sigma_2\sigma_3, \sigma_1\sigma_2\sigma_3\} \cong \mathbb{Z}_2 \times \mathbb{Z}_2 \times \mathbb{Z}_2
    \]
    \item We know that $K_i^* = \text{Gal}(E:K_i) = \{\sigma \in \text{aut}(E) : \sigma(u) = u \text{ for all } u \in K_i\}$. For $\sigma \in \text{aut}(E)$, $\sigma$ fixes $K_1$ if $\sigma(\sqrt{10}) = \sqrt{10}$. But $\sigma(\sqrt{10}) = \sigma(\sqrt{2} \sqrt{5}) = \sigma(\sqrt{2}) \sigma(\sqrt{5})$. So either $\sigma(\sqrt{2}) = -\sqrt{2}$ and $\sigma(\sqrt{5}) = -\sqrt{5}$, or $\sigma$ fixes both $\sqrt{2}$ and $\sqrt{5}$. Now $\text{irr}(\sqrt{10}, \mathbb{Q}) = x^2 - 10$ (irreducible by Eisenstein's Criterion), so a basis of $K_1$ over $\mathbb{Q}$ is $\{1, \sqrt{10}\}$. So fixing $\sqrt{10}$ is the only condition needed and thus \[
    K_1^* = \{\text{Id}, \sigma_2, \sigma_1 \sigma_3, \sigma_1 \sigma_2 \sigma_3\}
    \]
    For $\sigma \in K_2^*$, we must have that $\sigma (\sqrt{2} + \sqrt{5}) = \sqrt{2} + \sqrt{5}$. We know that $\sigma(\sqrt{2} + \sqrt{5}) = \sigma(\sqrt{2}) + \sigma(\sqrt{5}) = \pm \sqrt{2} \pm \sqrt{5}$. Thus $\sigma \in K_2^*$ must fix both $\sqrt{2}$ and $\sqrt{5}$. We must show this condition is enough to fix $K_2$. We have that \[
    \mathbb{Q}(\sqrt{2} + \sqrt{5}) \subseteq \mathbb{Q}(\sqrt{2}, \sqrt{5})
    \] 
    On the other hand, $(\sqrt{2} + \sqrt{5})^{-1} = \frac{\sqrt{5} - \sqrt{2}}{3} \in \mathbb{Q}(\sqrt{2} + \sqrt{5})$. So $\sqrt{2},\sqrt{5} \in \mathbb{Q}(\sqrt{2} + \sqrt{5})$ and thus $\mathbb{Q}(\sqrt{2} + \sqrt{5}) = \mathbb{Q}(\sqrt{2}, \sqrt{5})$. In the same way as we did in (a), a basis of $\mathbb{Q}(\sqrt{2}, \sqrt{5})$ over $\mathbb{Q}$ is $\{1, \sqrt{2}, \sqrt{5}, \sqrt{10}\}$. Thus fixing both $\sqrt{2}$ and $\sqrt{5}$ is enough to fix $K_1$ and so:
    \[
    K_2^* = \{\text{Id}, \sigma_2\}
    \]
    \item If $[K:\mathbb{Q}] = 4$, then, by the Degrees Law, $[E:\mathbb{Q}] = [E:K] \cdot [K:\mathbb{Q}]$ so $[E:K] = 2$. Now $E$ is the splitting field of the polynomial $f(x) = (x^2 - 2)(x^2 - 3) (x^2 - 5) \in \mathbb{Q}[x]$. So $E:\mathbb{Q}$ is normal. Now $\text{char}(\mathbb{Q}) = 0$, and $E:\mathbb{Q}$ is a finite normal extension, so it is a Galois finite extension. But then for any intermediate field $K$, $[E:K] = |K^*|$. So in order for $[K:\mathbb{Q}] = 4$, we need $[E:K] = |K^*| = 2$. Since $\text{Id} \in K^*$, $K^*$ contains exactly one other automorphism. Considering the 7 other automorphisms:
    \begin{align*}
      K & = \mathbb{Q}(\sqrt{3}, \sqrt{5}) \  (\text{fixed by }\sigma_1)\\
      K & = \mathbb{Q}(\sqrt{2}, \sqrt{5}) \  (\text{fixed by }\sigma_2)\\
      K & = \mathbb{Q}(\sqrt{2}, \sqrt{3})\  (\text{fixed by }\sigma_3)\\
      K & = \mathbb{Q}(\sqrt{6}, \sqrt{5}) \  (\text{fixed by }\sigma_1\sigma_2)\\
      K & = \mathbb{Q}(\sqrt{10}, \sqrt{3}) \  (\text{fixed by }\sigma_1\sigma_3)\\
      K & = \mathbb{Q}(\sqrt{15}, \sqrt{2})\  (\text{fixed by }\sigma_2\sigma_3)\\
      K & = \mathbb{Q}(\sqrt{6}, \sqrt{10})\  (\text{fixed by }\sigma_1\sigma_2\sigma_3)
    \end{align*}
    So there are 7 such fields.
    \item Let $u$ be a root of $f$ in some extension of $\mathbb{Q}$. Assume, to the contrary, that $u \in E$. Then \[
    \mathbb{Q} \subseteq \mathbb{Q}(u) \subseteq E
    \]
    But clearly $[\mathbb{Q}(u):\mathbb{Q}] = \text{deg}(u, \mathbb{Q}) = 3$, and an application of the degrees law gives us that \[
    [E: \mathbb{Q}] = [E:\mathbb{Q}(u)][\mathbb{Q}(u):\mathbb{Q}] \iff 8 = [E:\mathbb{Q}(u)] \cdot 3
    \]
    But then $[E:\mathbb{Q}(u)] = \frac{8}{3}$, and this is impossible since $[E:\mathbb{Q}(u)]$ must be an integer
\end{enumerate}
